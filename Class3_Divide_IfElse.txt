import { Component } from '@angular/core';
import { FormsModule } from '@angular/forms';

@Component({
  selector: 'app-root',
  imports: [FormsModule],
  template: `
    <h1>Hello, {{ name }}!!</h1>

    <p>Enter Number 1: <input type="number" [(ngModel)]="num1"></p>
    <p>Enter Number 2: <input type="number" [(ngModel)]="num2"></p>

    <button (click)="Divide2numbers()">Divide2numbers!</button>
  
    <h2>Total = {{ total }}</h2>
  `
})
export class App {
  name = 'Cathy the Angular Pro';
  num1 = 0;
  num2 = 0;
  total:any = 0;
 
 Divide2numbers() {
if (this.num2 ==0) 
this.total = "Denominator can't be zero";
else
{
    this.total = this.num1 / this.num2;
} 
}
}
