  isCaramelTheme: boolean = false;
  memberId: string = '';
  loyaltyMessage: string = '';
  cartCount: number = 0;
  cartTotal: number = 0;
  statusMessage: string = '';

  categoryTitle: string = '☕ Hot & Iced Coffees';

  // 1. ALL COFFEES
  coffeeList = [

  ];

  // 2. ALL CHAI & TEAS
  teaList = [
    { name: 'Masala Chai Tea Latte', desc: 'Black tea infused with cardamom, cinnamon, ginger & steamed milk', price: 5.25, calories: 240, icon: '🫖', warm: true, warmLabel: 'Extra Hot' },
    { name: 'Iced Dirty Chai Latte', desc: 'Spiced black tea blended with milk, ice & a bold shot of espresso', price: 5.65, calories: 260, icon: '🥤', warm: false, warmLabel: 'Light Ice' },
    { name: 'Matcha Green Tea Latte', desc: 'Smooth & creamy ceremonial matcha sweetened just right with milk', price: 5.45, calories: 240, icon: '🍵', warm: true, warmLabel: 'Extra Hot' },
    { name: 'London Fog Earl Grey', desc: 'Bright, citrusy Earl Grey tea with lavender & vanilla foam', price: 4.95, calories: 180, icon: '🌿', warm: true, warmLabel: 'Extra Hot' }
  ];

  // 3. ALL BAKERY, CAKES, BREAD & BISCUITS
  bakeryList = [
    { name: 'Iced Lemon Pound Cake', desc: 'Moist, zesty lemon cake topped with a sweet tart lemon icing', price: 4.25, calories: 470, icon: '🍋', warm: false, warmLabel: 'Warmed' },
    { name: 'Pistachio Swirl Loaf Cake', desc: 'Rich pistachio bakery cake with roasted pistachio crumble', price: 4.65, calories: 410, icon: '🍰', warm: true, warmLabel: 'Warmed' },
    { name: 'Banana Walnut Bread', desc: 'Home-style moist banana bread loaded with crunchy walnuts', price: 4.25, calories: 420, icon: '🍞', warm: true, warmLabel: 'Warmed' },
    { name: 'Golden Butter Tea Biscuits', desc: 'Crisp, buttery bakery biscuits — made for dipping in hot Chai!', price: 3.50, calories: 310, icon: '🍪', warm: true, warmLabel: 'Warmed' }
  ];

  // Starts by showing all Coffees!
  displayedItems = this.coffeeList;

  // Switches the grid when you click Coffee, Tea, or Bakery in the Side-Nav!
  showCategory(cat: string) {
    this.statusMessage = '';
    if (cat == 'Coffee') {
      this.categoryTitle = '☕ Hot & Iced Coffees';
      this.displayedItems = this.coffeeList;
    } else if (cat == 'Tea') {
      this.categoryTitle = '🫖 Chai & Artisanal Teas';
      this.displayedItems = this.teaList;
    } else if (cat == 'Bakery') {
      this.categoryTitle = '🍋 Bakery, Cakes, Bread & Tea Biscuits';
      this.displayedItems = this.bakeryList;
    }
  }

  checkLoyalty() {
    if (this.memberId == 'SBX100') {
      this.loyaltyMessage = '🌟 VIP Gold Member Cathy! 1,250 Stars — Free Bakery Item Ready!';
    } else if (this.memberId != '') {
      this.loyaltyMessage = '✅ Member #' + this.memberId + ' linked! Earning 2★ per $1.';
    } else {
      this.loyaltyMessage = '⚠️ Please enter your SBX Loyalty # first!';
    }
  }

  addToCart(item: any) {
    this.cartCount = this.cartCount + 1;
    this.cartTotal = this.cartTotal + item.price;
    this.statusMessage = '✅ Added ' + item.name + ' (' + (item.warm ? item.warmLabel : 'Standard') + ') to your SBX Bag!';
  }
}
