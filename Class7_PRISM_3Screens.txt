  plantLocation = '1007 - Colton';
  currentTab = 'Internal Sales Orders';
  message = '';

  // Data for Screen 1: Internal Sales Orders
  orderList: any = [

  ];

  // Data for Screen 2: Daily Bin Inventory
  binList: any = [

    { binName: '804', binType: 'WIP Storage', itemDesc: 'R54', itemNum: '2001438', feetDown: 38, inventory: 280,  emptyBin: false, capacity: 1800, openCapacity: 1520, binFeet: 45, cwtPerFoot: 40, status: 'Active' }
  ];

  // Data for Screen 3: Quality Hold
  holdType = 'Food Safety';
  holdStatus = 'On Hold';
  holdItemNum = '2000483';
  holdItemDesc = 'E50';
  holdList: any = [
    { logCode: 'QH-1001', status: 'On Hold', holdDate: '10/25/2026', type: 'Food Safety', itemNum: '2000483', itemDesc: 'E50' }
  ];

  SelectTab(tabName: any) {
    this.currentTab = tabName;
    this.message = '';
  }

  ReceiveOrder(ord: any) {
    if (ord.status == 'In Transit') {
      ord.status = 'Received';
      this.message = 'Order ' + ord.orderNum + ' has been Received!';
    } else {
      this.message = 'Cannot Receive Order ' + ord.orderNum + ' — Status is ' + ord.status;
    }
  }

  CalculateBin(bin: any) {
    if (bin.emptyBin == true) {
      bin.inventory = 0;
      bin.openCapacity = bin.capacity;
    } else {
      bin.openCapacity = bin.feetDown * bin.cwtPerFoot;
      bin.inventory = bin.capacity - bin.openCapacity;
    }
  }

  AddQualityHold() {
    this.holdList.push({
      logCode: 'QH-' + (1001 + this.holdList.length),
      status: this.holdStatus,
      holdDate: '10/25/2026',
      type: this.holdType,
      itemNum: this.holdItemNum,
      itemDesc: this.holdItemDesc
    });
    this.message = 'Added Quality Hold for Item ' + this.holdItemNum;
  }
}
